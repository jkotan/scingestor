def load_config(configfile):
    """ load config file

    :param configfile: configuration file name
    :type configfile: :obj:`str`
    """
    config = {}
    try:
        with open(configfile, 'r') as f:
            config = yaml.safe_load(f)
    except Exception as e:
        get_logger().warning(str(e))
    return config
